-- Criação das Tabelas
CREATE TABLE Cliente (
    codigo NUMBER PRIMARY KEY,
    nome VARCHAR2(100) NOT NULL,
    cidade VARCHAR2(100),
    idade NUMBER
);

CREATE TABLE Servicos (
    codigo NUMBER PRIMARY KEY,
    nome VARCHAR2(100) NOT NULL,
    valorreferencia NUMBER(10,2)
);

CREATE TABLE OrdemServico (
    numos NUMBER PRIMARY KEY,
    data DATE,
    valortotal NUMBER(10,2),
    cod_cliente NUMBER,
    CONSTRAINT fk_os_cliente FOREIGN KEY (cod_cliente) REFERENCES Cliente(codigo)
);

CREATE TABLE ItensOrdemServico (
    numitem NUMBER,
    numos NUMBER,
    codservico NUMBER,
    valor NUMBER(10,2),
    quantidade NUMBER,
    PRIMARY KEY (numitem, numos),
    CONSTRAINT fk_item_os FOREIGN KEY (numos) REFERENCES OrdemServico(numos),
    CONSTRAINT fk_item_servico FOREIGN KEY (codservico) REFERENCES Servicos(codigo)
);

-- Consultas
SELECT * FROM Cliente WHERE UPPER(cidade) = 'LINS';

SELECT * FROM Servicos;

SELECT c.codigo AS codigocliente, c.nome AS nomecliente, o.numos, o.valortotal
FROM Cliente c JOIN OrdemServico o ON c.codigo = o.cod_cliente;

SELECT c.codigo AS codigocliente, c.nome AS nomecliente, o.numos, o.valortotal, i.numitem, i.quantidade, s.nome AS nomeservico
FROM Cliente c JOIN OrdemServico o ON c.codigo = o.cod_cliente JOIN ItensOrdemServico i ON o.numos = i.numos JOIN Servicos s ON i.codservico = s.codigo;

SELECT c.codigo AS codigocliente, c.nome AS nomecliente, o.numos, i.numitem, i.quantidade, (i.quantidade * i.valor) AS valorpago, s.nome AS nomeservico
FROM Cliente c JOIN OrdemServico o ON c.codigo = o.cod_cliente JOIN ItensOrdemServico i ON o.numos = i.numos JOIN Servicos s ON i.codservico = s.codigo;

SELECT c.codigo AS codigocliente, c.nome AS nomecliente, o.numos, i.numitem, i.quantidade, (i.quantidade * i.valor) AS valorpago, s.nome AS nomeservico
FROM Cliente c JOIN OrdemServico o ON c.codigo = o.cod_cliente JOIN ItensOrdemServico i ON o.numos = i.numos JOIN Servicos s ON i.codservico = s.codigo
WHERE UPPER(c.cidade) = 'LINS';
